cmake_minimum_required(VERSION 3.24)
# RTX 5090 = Blackwell sm_120 (needs CUDA >= 12.8). Must be set BEFORE project().
set(CMAKE_CUDA_ARCHITECTURES 120 CACHE STRING "")
project(cuda_path_tracer LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(SANITIZE "" CACHE STRING "sanitizer (address,undefined | thread); tests only")

add_library(pt_flags INTERFACE)
target_compile_options(pt_flags INTERFACE -Wall -Wextra -fno-omit-frame-pointer)
if(CMAKE_BUILD_TYPE STREQUAL "Release")
  target_compile_options(pt_flags INTERFACE -O3 -march=native)
endif()
if(SANITIZE)
  target_compile_options(pt_flags INTERFACE -fsanitize=${SANITIZE} -g -O1)
  target_link_options(pt_flags INTERFACE -fsanitize=${SANITIZE})
endif()
include_directories(include)
find_package(Threads REQUIRED)

# CPU-only targets: always built
add_executable(pt_tests tests/test_main.cpp tests/test_geometry.cpp tests/test_bvh.cpp tests/test_render.cpp)
target_link_libraries(pt_tests PRIVATE pt_flags Threads::Threads)
add_executable(pt_render src/main_render.cpp)
target_link_libraries(pt_render PRIVATE pt_flags Threads::Threads)
add_executable(pt_bench src/main_bench.cpp)
target_link_libraries(pt_bench PRIVATE pt_flags Threads::Threads)

# CUDA targets: only if a CUDA toolchain is found. Pass -DCMAKE_CUDA_COMPILER=/usr/local/cuda-12.8/bin/nvcc
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  add_library(pt_gpu STATIC cuda/gpu_renderer.cu)
  # -lineinfo for Nsight Compute; -Xptxas=-v prints register/spill info; no fast-math so CPU/GPU stay comparable
  target_compile_options(pt_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo -Xptxas=-v --expt-relaxed-constexpr>)
  target_compile_definitions(pt_gpu PUBLIC PT_HAS_CUDA)
  foreach(t pt_render pt_bench)
    target_link_libraries(${t} PRIVATE pt_gpu)
    target_compile_definitions(${t} PRIVATE PT_HAS_CUDA)
  endforeach()
  message(STATUS "CUDA found: GPU renderer enabled")
else()
  message(STATUS "CUDA not found: building CPU-only")
endif()
